function normalizeHeader(text) { return String(text || '').replace(/\s+/g, ' ').trim().toUpperCase(); } /** * Converts a raw cell value to a clean, readable date label. Google Sheets * sometimes returns date-like header cells as real JS Date objects rather * than strings -- this formats those consistently as "M/d" instead of * leaking a full timestamp string into the dashboard. */ function formatDateLabel(rawVal) { if (rawVal instanceof Date) { return Utilities.formatDate(rawVal, Session.getScriptTimeZone(), 'M/d'); } return String(rawVal || '').trim(); } /** * Parses the wide PR + per-session block on a lift data sheet/tab. * Works for both the current summer tab and the archive tab, since both * follow the same general pattern: a date-block header row, a metric-label * row beneath it, then athlete rows. Also captures "Reps" columns that * immediately follow a weight metric column (Bench/Squat/Deadlift), so * each lift can be paired with how many reps were completed. * * firstNameCol/lastNameCol let callers specify which columns hold first/last * name, since this differs between the two source sheets. * * PR columns are only recognized in the region BEFORE the first * dated session block begins (i.e. before the date row has any non-empty * cell) -- this avoids accidentally treating a later date-block's "Bench" * or "MPH" column as if it were the PR column. */ function buildLiftColumnMap(sheet, firstNameCol, lastNameCol) { const data = sheet.getDataRange().getValues(); let dateRowIdx = -1, metricRowIdx = -1; for (let r = 0; r < data.length; r++) { const rowText = data[r].join(' '); if (/Tuesday|Wednesday|Thursday/i.test(rowText)) { dateRowIdx = r; metricRowIdx = r + 1; r++) { const rowText = data[r].join(' '); if (/Reps/i.test(rowText) && /\d{1,2}\/\d{1,2}/.test(rowText)) { dateRowIdx = r; metricRowIdx = r + 1; break; } } } if (dateRowIdx === -1) return null; const dateRow = data[dateRowIdx]; const metricRow = data[metricRowIdx]; const dateLabels = []; let lastDate = ''; for (let c = 0; c < dateRow.length; c++) { const label = formatDateLabel(dateRow[c]); if (label) lastDate = label; dateLabels.push(lastDate); } const columnMap = []; // { col, date, metric, repsCol } const prColumnMap = {}; // metricKey -> col index const usedDateCols = new Set(); // prevents duplicate matches on same column for same metric for (let c = 0; c < metricRow.length; c++) { const header = normalizeHeader(metricRow[c]); if (!header) continue; // A column belongs to the PR region if its forward-filled date label // is literally "PRs" (case-insensitive), OR if no date label has // appeared yet at all (i.e. it's before the very first dated block). const dateLabelHere = normalizeHeader(dateLabels[c]); // already found if (variants.some(v => header === normalizeHeader(v))) { prColumnMap[metricKey] = c; break; } } continue; // PR-region columns are never treated as dated session data } for (const [metricKey, variants] of Object.entries(METRIC_HEADERS)) { const colKey = metricKey + '|' + dateLabels[c]; if (usedDateCols.has(colKey)) continue; // skip duplicate metric+date pairing if (variants.some(v => header === normalizeHeader(v))) { let repsCol = -1; if (c + 1 < metricRow.length && normalizeHeader(metricRow[c + 1]) === 'REPS') { repsCol = c + 1; } columnMap.push({ col: c, date: dateLabels[c], metric: metricKey, repsCol: repsCol }); usedDateCols.add(colKey); break; } } } return { dataStartRow: metricRowIdx + 1, columnMap: columnMap, prColumnMap: prColumnMap, rawData: data, firstNameCol: firstNameCol, lastNameCol: lastNameCol }; } function findAthleteRow(map, firstName, lastName) { for (let r = map.dataStartRow; r < map.rawData.length; r++) { const rowFirst = String(map.rawData[r][map.firstNameCol] || '').trim().toLowerCase(); const rowLast = String(map.rawData[r][map.lastNameCol] || '').trim().toLowerCase(); if (rowFirst === firstName.toLowerCase() && rowLast === lastName.toLowerCase()) { return r; } } return -1; } function isUsableValue(rawVal) { if (rawVal === '' || rawVal === null || rawVal === undefined) return false; if (typeof rawVal === 'string') { const v = rawVal.trim().toLowerCase(); if (v === 'x' || v === 'p' || v.indexOf('#') === 0) return false; } return true; } /** * Extracts { series: {...}, prs: {...} } for one athlete from a single * parsed sheet map. series values include reps when a paired Reps column * was found next to that metric's column. */ function extractAthleteData(map, rowIdx) { const row = map.rawData[rowIdx]; keys.forEach(key => { const a = (archiveSeries && archiveSeries[key]) || []; const s = (summerSeries && summerSeries[key]) || []; merged[key] = a.concat(s); }); return merged; } /** * PRs: summer sheet's PR wins if present; } /** * Reads the Check-In Log tab and returns attendance stats for one athlete: * - sessions: array of { date, quarter, points } for all check-ins * - byQuarter: { Q2: { attended, total, pct }, Q3: {...}, Q4: {...} } * - streak: current consecutive sessions attended (0 if last session missed) * Total possible sessions per quarter pulled from QUARTER_SCHEDULE. */ function getAthleteAttendance(athleteName) { try { const ss = SpreadsheetApp.getActiveSpreadsheet();
